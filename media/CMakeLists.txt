# Picture decoding and review-movie export. FFmpeg is optional: without it
# the decoder and encoder are stubs that say so, and the rest of picture
# support (timecode, markers, cues) works.
#
# Found through pkg-config, or - for a prebuilt LGPL shared FFmpeg with no
# usable .pc files (Windows release builds) - from VCC_FFMPEG_ROOT, a folder
# holding include/ and lib/ (and bin/ with the DLLs, which installers copy).
option(VCC_WITH_FFMPEG "Decode video with FFmpeg when it is available" ON)
set(VCC_FFMPEG_ROOT "" CACHE PATH "A prebuilt FFmpeg (include/, lib/, bin/) when pkg-config cannot find one")
vcc_add_library(vcc_media)
set(VCC_HAVE_FFMPEG OFF PARENT_SCOPE)
if(VCC_WITH_FFMPEG)
    set(_vcc_ffmpeg_target "")
    if(VCC_FFMPEG_ROOT)
        find_path(VCC_FFMPEG_INCLUDE libavcodec/avcodec.h PATHS "${VCC_FFMPEG_ROOT}/include" NO_DEFAULT_PATH)
        set(_vcc_ffmpeg_libs "")
        foreach(_lib avformat avcodec swscale avutil)
            find_library(VCC_FFMPEG_${_lib} NAMES ${_lib} PATHS "${VCC_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH)
            if(VCC_FFMPEG_${_lib})
                list(APPEND _vcc_ffmpeg_libs "${VCC_FFMPEG_${_lib}}")
            endif()
        endforeach()
        list(LENGTH _vcc_ffmpeg_libs _vcc_ffmpeg_count)
        if(VCC_FFMPEG_INCLUDE AND _vcc_ffmpeg_count EQUAL 4)
            add_library(vcc_ffmpeg INTERFACE)
            target_include_directories(vcc_ffmpeg SYSTEM INTERFACE "${VCC_FFMPEG_INCLUDE}")
            target_link_libraries(vcc_ffmpeg INTERFACE ${_vcc_ffmpeg_libs})
            set(_vcc_ffmpeg_target vcc_ffmpeg)
            message(STATUS "Video: FFmpeg from ${VCC_FFMPEG_ROOT}")
        else()
            message(WARNING "VCC_FFMPEG_ROOT=${VCC_FFMPEG_ROOT} has no complete FFmpeg (include/ and avformat, avcodec, swscale, avutil in lib/)")
        endif()
    endif()
    if(NOT _vcc_ffmpeg_target)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(FFMPEG IMPORTED_TARGET libavformat libavcodec libswscale libavutil)
        endif()
        if(FFMPEG_FOUND)
            set(_vcc_ffmpeg_target PkgConfig::FFMPEG)
            message(STATUS "Video: FFmpeg ${FFMPEG_libavformat_VERSION}")
        endif()
    endif()
    if(_vcc_ffmpeg_target)
        target_link_libraries(vcc_media PRIVATE ${_vcc_ffmpeg_target})
        target_compile_definitions(vcc_media PRIVATE VCC_WITH_FFMPEG=1)
        set(VCC_HAVE_FFMPEG ON PARENT_SCOPE)
    else()
        message(STATUS "Video: FFmpeg not found - picture import and export disabled in this build")
    endif()
endif()
