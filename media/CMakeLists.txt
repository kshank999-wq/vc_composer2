# Picture decoding. FFmpeg is optional: without it the decoder is a stub that
# says so, and the rest of picture support (timecode, markers, cues) works.
option(VCC_WITH_FFMPEG "Decode video with FFmpeg when it is available" ON)
vcc_add_library(vcc_media)
set(VCC_HAVE_FFMPEG OFF PARENT_SCOPE)
if(VCC_WITH_FFMPEG)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(FFMPEG IMPORTED_TARGET libavformat libavcodec libswscale libavutil)
    endif()
    if(FFMPEG_FOUND)
        target_link_libraries(vcc_media PRIVATE PkgConfig::FFMPEG)
        target_compile_definitions(vcc_media PRIVATE VCC_WITH_FFMPEG=1)
        set(VCC_HAVE_FFMPEG ON PARENT_SCOPE)
        message(STATUS "Video decoding: FFmpeg ${FFMPEG_libavformat_VERSION}")
    else()
        message(STATUS "Video decoding: FFmpeg not found - picture import disabled in this build")
    endif()
endif()
