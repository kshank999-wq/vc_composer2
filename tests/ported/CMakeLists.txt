# Suites ported with the original VC Composer modules. They keep their own
# harness (support/vcc_test.h: VCC_TEST / VCC_CHECK), which is why they live
# beside, not inside, tests/unit.
function(vcc_add_ported_test name)
    add_executable(${name} unit/${name}.cpp)
    target_include_directories(${name} PRIVATE support)
    target_link_libraries(${name} PRIVATE ${ARGN} vcc_build_options)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
endfunction()
